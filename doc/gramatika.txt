<program> → import ifj26.* EOL <func_list>
<func_list> → <func_def> <func_list>
<func_list> → epsilon
<func_def> → <opt_main> def id ( <param_list> ) <opt_type> = <func_body>

<opt_main> → @main <opt_main_nl>
<opt_main> → epsilon
<opt_main_nl> → EOL
<opt_main_nl> → epsilon
<opt_type> → : <type>
<opt_type> → epsilon
<type> → Int
<type> → Double
<type> → String
<type> → Unit
<param_list> → <param> <param_next>
<param_list> → epsilon
<param> → id : <type>
<param_next> → , <param_next>
<param_next> → epsilon

<func_body> → <expression_or_func_call> EOL
<func_body> → <block>

<block> → EOL TOKEN_INDENT <statement_list> TOKEN_DEDENT
<statement_list> → <statement> <statement_list_n>
<statement_list_n> → <statement> <statement_list_n>
<statement_list_n> → epsilon

<statement> → <var_decl> EOL
<statement> → <assign_statement> EOL
<statement> → <return_statement> EOL
<statement> → <expression> EOL
<statement> → <func_call> EOL

<statement> → <if_statement>
<statement> → <while_statement>

<if_statement> → if <expression> then <block> else <block>
<while_statement> → while <expression> do <block>

<var_decl> → val id <opt_type> = <expression_or_func_call>
<var_decl> → var id <opt_type> = <expression_or_func_call>
<assign_statement> → id = <expression_or_func_call>
<return_statement> → return <opt_value>
<opt_value> → <expression_or_func_call>
<opt_value> → epsilon

<expression_or_func_call> → <expression>
<expression_or_func_call> → <func_call>

<func_call> → id ( <arg_list> )
<arg_list> → epsilon
<arg_list> → <term> <arg_next>
<arg_next> → , <arg_next2>
<arg_next> → epsilon
<arg_next2> → <term> <arg_next>
<arg_next2> → epsilon

<term> → TOKEN_LITERAL_INT
<term> → TOKEN_LITERAL_STRING
<term> → TOKEN_LITERAL_DOUBLE
<term> → TOKEN_LITERAL_UNIT
<term> → id
